function startSparkListening(){if(!(window.SparkAge&&SparkAge.prefs&&SparkAge.prefs().mic))return;const SR=window.SpeechRecognition||window.webkitSpeechRecognition;if(!SR){setSparkState('unsure','Voice input unavailable');speak('I can still explore with you. Tap something in the world and I will help.',{after:'unsure'});return}try{recognition?.abort()}catch{};recognition=new SR();recognition.lang='en-US';recognition.interimResults=false;recognition.maxAlternatives=1;recognition.onstart=()=>setSparkState('listening');recognition.onresult=e=>{const text=e.results?.[0]?.[0]?.transcript||'';updateSparkContext({lastChildSpeech:text});setSparkState('thinking');setTimeout(()=>respondToChild(text),250)};recognition.onerror=()=>setSparkState('unsure');recognition.onend=()=>{if(sparkState==='listening')setSparkState('idle')};recognition.start()}

function baseReply(raw=''){const text=String(raw).trim().toLowerCase();if(!text)return speak(`We are exploring ${contextPlace()}. function numberChoices(answer,spread=2){const n=Number(answer),values=[n,n+spread,Math.max(0,n-spread),n+spread*2];return[...new Set(values)].map(String)}
function firstLetterChoices(name){const a=name[0].toUpperCase(),pool=[a,'M','S','E','J','N','V','U'].filter((x,i,p)=>p.indexOf(x)===i);return[a,...pool.filter(x=>x!==a).slice(0,3)]}
